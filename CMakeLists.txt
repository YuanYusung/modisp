cmake_minimum_required(VERSION 3.22)
project(modisp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
pybind11_add_module(_cpu NO_EXTRAS cpp/cpu/solver.cpp cpp/cpu/dunkin_ref.cpp)
target_include_directories(_cpu PRIVATE cpp)
target_link_libraries(_cpu PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(_cpu PRIVATE /O2)
else()
  target_compile_options(_cpu PRIVATE -O3 -ffp-contract=off)
endif()
option(MD_SANITIZE "Enable CPU address/undefined behavior sanitizers for testing" OFF)
if(MD_SANITIZE AND NOT MSVC)
  target_compile_options(_cpu PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(_cpu PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS _cpu LIBRARY DESTINATION modisp)
